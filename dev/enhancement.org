  + =state= = describe decision; =approved=, =blocked=, =inactive= or =pending=
  + =type= = describe type; =bug=, =chore=, =discussion=, =docs=, =feature=, =fix=, =security=, =testing=
  + =work= = describe situation; =chaotic=, =complex=, =complicated= or =obvious=
